function Hash([string]$path) { (Get-FileHash -LiteralPath $path -Algorithm SHA256).Hash.ToLowerInvariant() }
function WriteNew([string]$name, [byte[]]$bytes) {
    $stream = [IO.File]::Open((Join-Path $PSScriptRoot $name), [IO.FileMode]::CreateNew, [IO.FileAccess]::Write, [IO.FileShare]::Read)
    try { $stream.Write($bytes, 0, $bytes.Length); $stream.Flush($true) } finally { $stream.Dispose() }
}
function JsonNew([string]$name, $value) {
    WriteNew $name ([Text.Encoding]::UTF8.GetBytes(($value | ConvertTo-Json -Depth 12 -Compress)))
}
function RunPinned([string]$arguments) {
    $start = New-Object Diagnostics.ProcessStartInfo
    $start.FileName = $dotnet; $start.Arguments = $arguments
    $start.UseShellExecute = $false; $start.CreateNoWindow = $true
    $start.RedirectStandardOutput = $true; $start.RedirectStandardError = $true
    foreach ($name in @($start.EnvironmentVariables.Keys)) {
        if ($name -match '^(DOTNET_|COMPLUS_|CORECLR_|COR_|COREHOST_)' -or
            $name -in @('APPDOMAIN_MANAGER_ASM', 'APPDOMAIN_MANAGER_TYPE', 'DEVPATH', '__COMPAT_LAYER')) {
            $start.EnvironmentVariables.Remove($name)
        }
    }
    $child = [Diagnostics.Process]::Start($start)
    try {
        $stdout = $child.StandardOutput.ReadToEndAsync(); $stderr = $child.StandardError.ReadToEndAsync()
        if (-not $child.WaitForExit(30000)) { throw 'PROBE_CHILD_UNRESOLVED_DO_NOT_RETRY' }
        if ($stdout.Result.Length -gt 65536 -or $stderr.Result.Length -gt 65536) { throw 'PROBE_OUTPUT_BOUND_REFUSED' }
        [pscustomobject]@{ exitCode = $child.ExitCode; stdout = $stdout.Result; stderr = $stderr.Result }
    } finally { $child.Dispose() }
}
